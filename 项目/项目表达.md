public class TokenBucket {  
 String key;  
 Integer rate = 5;//新的一秒生成多少令牌  
 Integer capacity = 5;  
 Integer tokens = 1;  
// Long now = System.currentTimeMillis();//Lua脚本根据不断传入的当前时间判断令牌桶中的数量  
}


private static final String SCRIPT =  
        "local key = KEYS[1] " +  
                "local current_tokens_key = key .. ':tokens' " +  
                "local refill_time_key = key .. ':last_refill_timestamp' " +  
                "local rate = tonumber(ARGV[1]) " +  
                "local capacity = tonumber(ARGV[2]) " +  
                "local redis_time = redis.call('TIME') " +  
                "local now = tonumber(redis_time[1]) + tonumber(redis_time[2]) / 1000000 " +  
                "local requested_tokens = tonumber(ARGV[3]) " +  
                "local current_tokens = tonumber(redis.call('get', current_tokens_key) or capacity) " +  
                "local last_refill_timestamp = tonumber(redis.call('get', refill_time_key) or now) " +  
                "local delta_time = now - last_refill_timestamp " +  
                "local new_tokens = math.min(capacity, current_tokens + (delta_time * rate)) " +  
                "redis.call('set', refill_time_key, now) " +  
                "if new_tokens >= requested_tokens then " +  
                " new_tokens = new_tokens - requested_tokens " +  
                " redis.call('set', current_tokens_key, new_tokens) " +  
                " return 1 " +  // 返回 1 表示成功  
                "else " +  
                " redis.call('set', current_tokens_key, new_tokens) " +  
                " return 0 " +  // 返回 0 表示失败  
                "end";


#### 预警模块
“我负责的智慧工地模块需要对接海康ISC平台接收预警。海康的预警推送有一个特点——突发性很强，比如某个工人误入禁区，会在短时间内收到大量重复预警。而且预警后我们要同时做三件事：存库、发站内信、发短信通知。

最初是同步串行处理，接口响应要800多毫秒，而且短信接口偶尔超时会导致整个预警处理失败。

后来我引入了RabbitMQ做改造：

海康预警进来后直接发到MQ，立即返回200给海康，响应时间降到120ms

使用Topic Exchange把预警同时路由到三个队列：落库、站内信、短信

落库队列用手动ACK，确保消息不丢

短信队列配置重试机制，失败后自动重试3次，间隔5秒，仍然失败进入死信队列，由运维人工兜底

为了防止同一个预警重复处理，我还结合Redis存储了预警ID的幂等性校验

改造后，短信接口故障完全不影响预警核心功能，系统稳定性大大提升。”
